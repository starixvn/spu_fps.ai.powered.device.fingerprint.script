class FingerprintEngine {
    constructor() {
        this.murmur3 = this.initMurmur3();
    }

    // Main fingerprint collection with immediate private data submission
    async getStructuredFingerprint() {
        try {
            const groups = await Promise.all([
                this.collectGroupA_Hardware(),
                this.collectGroupB_Graphics(),
                this.collectGroupC_Audio(),
                this.collectGroupD_Browser(),
                this.collectGroupE_APIs()
            ]);

            const fullFingerprint = {
                sessionId: this.generateSessionId(),
                deviceUUID: await this.getDeviceUUID(),
                timestamp: new Date().toISOString(),
                groups: {
                    groupA_hardware: groups[0],
                    groupB_graphics: groups[1],
                    groupC_audio: groups[2],
                    groupD_browser: groups[3],
                    groupE_apis: groups[4]
                },
                webrtcIPs: await this.getWebRTCIPs(),
                riskFactors: await this.detectRiskFactors(),
                finalHash: this.generateFinalHash(groups)
            };

            // Separate data for privacy-first approach
            const separatedData = this.separateDataForPrivacy(fullFingerprint);
            
            // IMMEDIATE private data submission - don't wait for user action
            console.log('[FingerprintEngine] Immediately submitting private data after collection');
            this.submitPrivateDataImmediately(separatedData.privateData);
            
            return separatedData;
        } catch (error) {
            console.warn('[FingerprintEngine] Collection failed:', error);
            return this.getFallbackFingerprint();
        }
    }

    // Single private data submission when collection completes
    async submitPrivateDataImmediately(privateData) {
        console.log(`[FingerprintEngine] Submitting private data for session: ${privateData.sessionId}`);
        
        // Single submission attempt when fingerprint collection is successful
        this.submitPrivateDataViaBeacon(privateData, 1);
        
        // Store for page lifecycle events (additional safety net)
        this.storedPrivateData = privateData;
        this.setupPageLifecycleSubmission();
    }

    // Enhanced Beacon API submission with attempt tracking
    async submitPrivateDataViaBeacon(privateData, attemptNumber = 1) {
        // Check if already successfully submitted
        if (this.privateDataSubmitted) {
            console.log('[FingerprintEngine] Private data already submitted successfully');
            return true;
        }

        try {
            console.log(`[FingerprintEngine] Private data submission attempt ${attemptNumber} for ${privateData.sessionId}`);
            
            // Enhanced payload with submission metadata
            const enhancedPayload = {
                ...privateData,
                submissionMeta: {
                    attempt: attemptNumber,
                    submissionTime: new Date().toISOString(),
                    timeSinceCollection: Date.now() - new Date(privateData.timestamp).getTime(),
                    connectionType: navigator.connection?.effectiveType || 'unknown',
                    isImmediate: true
                }
            };

            const payload = JSON.stringify(enhancedPayload);
            
            // Primary: Beacon API for guaranteed delivery
            if (navigator.sendBeacon) {
                const blob = new Blob([payload], { type: 'application/json' });
                const success = navigator.sendBeacon('/fps/v1/device-details', blob);
                if (success) {
                    console.log(`[FingerprintEngine] ✅ Private data submitted via Beacon API (attempt ${attemptNumber})`);
                    this.privateDataSubmitted = true;
                    this.cleanupPrivateDataStorage();
                    return true;
                }
            }

            // Fallback: Fetch with keepalive
            const response = await fetch('/fps/v1/device-details', {
                method: 'POST',
                body: payload,
                headers: { 
                    'Content-Type': 'application/json',
                    'X-FPS-Attempt': attemptNumber.toString(),
                    'X-FPS-Immediate': 'true'
                },
                keepalive: true
            });

            if (response.ok) {
                console.log(`[FingerprintEngine] ✅ Private data submitted via Fetch keepalive (attempt ${attemptNumber})`);
                this.privateDataSubmitted = true;
                this.cleanupPrivateDataStorage();
                return true;
            }
            
        } catch (error) {
            console.warn(`[FingerprintEngine] Private data submission attempt ${attemptNumber} failed:`, error);
        }

        return false;
    }

    // Setup page lifecycle submission as additional safety net
    setupPageLifecycleSubmission() {
        // Only setup if not already done
        if (this.lifecycleHandlersSetup) return;
        this.lifecycleHandlersSetup = true;

        // Submit on page visibility change (user switching tabs)
        document.addEventListener('visibilitychange', () => {
            if (document.visibilityState === 'hidden' && !this.privateDataSubmitted && this.storedPrivateData) {
                console.log('[FingerprintEngine] Page hidden - final private data submission attempt');
                this.submitPrivateDataViaBeacon(this.storedPrivateData, 'lifecycle-hidden');
            }
        });

        // Submit on beforeunload (user closing/navigating away)
        window.addEventListener('beforeunload', () => {
            if (!this.privateDataSubmitted && this.storedPrivateData) {
                console.log('[FingerprintEngine] Page unloading - critical private data submission');
                this.submitPrivateDataViaBeacon(this.storedPrivateData, 'lifecycle-unload');
            }
        });

        // Submit on page freeze (mobile browsers)
        document.addEventListener('freeze', () => {
            if (!this.privateDataSubmitted && this.storedPrivateData) {
                this.submitPrivateDataViaBeacon(this.storedPrivateData, 'lifecycle-freeze');
            }
        });
    }

    // Cleanup after successful submission
    cleanupPrivateDataStorage() {
        this.storedPrivateData = null;
        try {
            sessionStorage.setItem('fps_private_submitted', 'true');
        } catch (e) {
            // Ignore storage errors
        }
    }

    // Separate sensitive data from public data
    separateDataForPrivacy(fullFingerprint) {
        const publicData = {
            sessionId: fullFingerprint.sessionId,
            deviceUUID: fullFingerprint.deviceUUID,
            timestamp: fullFingerprint.timestamp,
            basicFingerprint: {
                hardwareHash: fullFingerprint.groups.groupA_hardware.hash,
                graphicsHash: fullFingerprint.groups.groupB_graphics.hash,
                audioHash: fullFingerprint.groups.groupC_audio.hash,
                browserHash: fullFingerprint.groups.groupD_browser.hash,
                apiHash: fullFingerprint.groups.groupE_apis.hash
            },
            confidence: this.calculateOverallConfidence(fullFingerprint.groups),
            privacy: "minimal_data_collection"
        };

        const privateData = {
            sessionId: fullFingerprint.sessionId, // Linking key
            timestamp: fullFingerprint.timestamp,
            detailedGroups: fullFingerprint.groups,
            riskFactors: fullFingerprint.riskFactors,
            webrtcIPs: fullFingerprint.webrtcIPs,
            privacyLevel: "detailed_analysis"
        };

        return { publicData, privateData };
    }

    // Submit private data using Beacon API for guaranteed delivery
    async submitPrivateDataViaBeacon(privateData) {
        try {
            const payload = JSON.stringify(privateData);
            
            // Primary: Beacon API for guaranteed delivery
            if (navigator.sendBeacon) {
                const success = navigator.sendBeacon('/fps/v1/device-details', payload);
                if (success) {

                method: 'POST',
                body: payload,
                headers: { 'Content-Type': 'application/json' },
                keepalive: true // Survives page closure
            });
            
            console.log('[FPS] Private data submitted via Fetch keepalive');
            return true;
            
        } catch (error) {
            console.warn('[FPS] Private data submission failed:', error);
            return false;
        }
    }

    // Helper methods for specific attribute collection
    getArchitecture() {
        const canvas = document.createElement('canvas');
        const gl = canvas.getContext('webgl') || canvas.getContext('experimental-webgl');
        if (gl) {

        totalScore += factors[factor] * weight * 100;
    }
    
    const finalScore = Math.min(Math.max(Math.round(totalScore), 0), 100);
    
    return {
        score: finalScore,
        factors: factors,
        processingTime: Date.now() - startTime,
        confidence: calculateConfidence(factors)
    };
}

WITH user_device_history AS (
    SELECT 
        us.user_id,

        us.fingerprint_data,
        us.client_ip,
        us.created_at,
        LAG(us.fingerprint_data) OVER (
            PARTITION BY us.user_id 
            ORDER BY us.created_at
        ) as prev_fingerprint,
        ROW_NUMBER() OVER (
            PARTITION BY us.user_id 
            ORDER BY us.created_at DESC
        ) as session_rank
    FROM user_sessions us
    WHERE us.user_id = ? 
    AND us.created_at > NOW() - INTERVAL '30 days'
),
device_consistency_metrics AS (
    SELECT 
        user_id,
        COUNT(*) as total_sessions,
        COUNT(DISTINCT fingerprint_data::text) as unique_fingerprints,
        COUNT(CASE 
            WHEN fingerprint_data::text != prev_fingerprint::text 
            THEN 1 END
        ) as fingerprint_changes,
        AVG(CASE 
            WHEN fingerprint_data::text = prev_fingerprint::text 
            THEN 1.0 ELSE 0.0 END
        ) as consistency_rate
    FROM user_device_history 
    WHERE session_rank <= 100  -- Only last 100 sessions
    GROUP BY user_id
),
fraud_similarity_check AS (
    SELECT 
        udh.user_id,
        MAX(CASE 
            WHEN fp.fraud_indicator = true 
            THEN calculate_fingerprint_similarity(
                udh.fingerprint_data, fp.fingerprint_pattern
            )
            ELSE 0 
        END) as max_fraud_similarity
    FROM user_device_history udh
    LEFT JOIN fraud_patterns fp ON true
    WHERE udh.session_rank = 1
    GROUP BY udh.user_id
)
SELECT 
    dcm.user_id,
    
    -- Final weighted risk score (0-100)
    ROUND(
        (CASE 
            WHEN dcm.consistency_rate > 0.95 THEN 0.05
            WHEN dcm.consistency_rate > 0.85 THEN 0.15
            WHEN dcm.consistency_rate > 0.70 THEN 0.30
            WHEN dcm.consistency_rate > 0.50 THEN 0.60
            ELSE 0.85
        END * 0.30 +  -- Device consistency (30%)
        
        CASE 
            WHEN fsc.max_fraud_similarity > 0.95 THEN 0.90
            WHEN fsc.max_fraud_similarity > 0.85 THEN 0.75
            WHEN fsc.max_fraud_similarity > 0.70 THEN 0.40
            ELSE 0.05
        END * 0.25)  -- Fingerprint similarity (25%)
        * 100
    ) as final_risk_score,
    
    -- Risk category
    CASE 
        WHEN final_risk_score BETWEEN 0 AND 20 THEN 'Very Low Risk'
        WHEN final_risk_score BETWEEN 21 AND 40 THEN 'Low Risk'
        WHEN final_risk_score BETWEEN 41 AND 60 THEN 'Medium Risk'
        WHEN final_risk_score BETWEEN 61 AND 80 THEN 'High Risk'
        ELSE 'Very High Risk'
    END as risk_category,
    
    dcm.consistency_rate,
    fsc.max_fraud_similarity,
    NOW() as calculated_at

FROM device_consistency_metrics dcm
LEFT JOIN fraud_similarity_check fsc ON dcm.user_id = fsc.user_id;

    constructor() {

    // Main initialization - starts immediately when script loads
    async initialize() {
        if (this.collectionStarted) {
            console.log('[FPS Core] Collection already started, returning existing promise');
            return this.publicDataPromise;
        }

        this.collectionStarted = true;
        console.log('[FPS Core] Starting background fingerprint collection...');
        
        try {
            // Start complete fingerprint collection
            const fingerprintData = await this.fingerprintEngine.getStructuredFingerprint();
            
            // Separate data for privacy-first approach
            const separatedData = this.fingerprintEngine.separateDataForPrivacy(fingerprintData);
            
            // IMMEDIATE private data submission via Beacon API
            console.log('[FPS Core] 🚀 Submitting private data immediately...');
            this.submitPrivateDataImmediately(separatedData.privateData);
            
            // Store public data promise for login integration
            this.publicDataPromise = Promise.resolve(separatedData.publicData);
            
            console.log('[FPS Core] ✅ Fingerprint collection completed');
            console.log('[FPS Core] ✅ Private data submission initiated');
            console.log('[FPS Core] ⏳ Public data ready for login submission');
            
            return separatedData.publicData;
            
        } catch (error) {
            console.warn('[FPS Core] Collection failed:', error);
            const fallbackData = this.getFallbackPublicData();
            this.publicDataPromise = Promise.resolve(fallbackData);
            return fallbackData;
        }
    }

    // Single private data submission when collection completes
    async submitPrivateDataImmediately(privateData) {
        console.log(`[FPS Core] Submitting private data for session: ${privateData.sessionId}`);
        
        // Single submission when fingerprint is ready
        this.submitViaBeacon(privateData, 1);
    }

    // Beacon API submission
    async submitViaBeacon(privateData, attemptNumber = 1) {
        if (this.privateDataSubmitted) {
            console.log('[FPS Core] Private data already submitted successfully');
            return true;
        }

        try {
            console.log(`[FPS Core] Submitting private data via Beacon API`);
            
            const enhancedPayload = {
                ...privateData,
                submissionMeta: {
                    attempt: attemptNumber,
                    submissionTime: new Date().toISOString(),
                    timeSinceCollection: Date.now() - new Date(privateData.timestamp).getTime(),
                    connectionType: navigator.connection?.effectiveType || 'unknown',
                    method: 'immediate_beacon'
                }
            };

            const payload = JSON.stringify(enhancedPayload);
            
            // Primary: Beacon API for guaranteed delivery
            if (navigator.sendBeacon) {
                const blob = new Blob([payload], { type: 'application/json' });
                const success = navigator.sendBeacon('/fps/v1/device-details', blob);
                if (success) {
                    console.log(`[FPS Core] ✅ Private data submitted via Beacon API (attempt ${attemptNumber})`);
                    this.privateDataSubmitted = true;
                    this.notifySubmissionSuccess();
                    return true;
                }
            }

            // Fallback: Fetch with keepalive
            const response = await fetch('/fps/v1/device-details', {
                method: 'POST',
                body: payload,
                headers: { 
                    'Content-Type': 'application/json',
                    'X-FPS-Attempt': attemptNumber.toString(),
                    'X-FPS-Method': 'immediate'
                },
                keepalive: true
            });

            if (response.ok) {
                console.log(`[FPS Core] ✅ Private data submitted via Fetch keepalive (attempt ${attemptNumber})`);
                this.privateDataSubmitted = true;
                this.notifySubmissionSuccess();
                return true;
            }
            
        } catch (error) {
            console.warn(`[FPS Core] Private data submission attempt ${attemptNumber} failed:`, error);
        }

        return false;
    }

    // Setup page lifecycle handlers for additional reliability
    setupPageLifecycleSubmission(privateData) {
        // Submit on page visibility change
        document.addEventListener('visibilitychange', () => {
            if (document.visibilityState === 'hidden' && !this.privateDataSubmitted) {
                console.log('[FPS Core] Page hidden - final private data submission');
                this.submitViaBeacon(privateData, 'lifecycle-hidden');
            }
        });

        // Submit on beforeunload
        window.addEventListener('beforeunload', () => {
            if (!this.privateDataSubmitted) {
                console.log('[FPS Core] Page unloading - critical private data submission');
                this.submitViaBeacon(privateData, 'lifecycle-unload');
            }
        });

        // Submit on page freeze (mobile)
        document.addEventListener('freeze', () => {
            if (!this.privateDataSubmitted) {
                this.submitViaBeacon(privateData, 'lifecycle-freeze');
            }
        });
    }

    // Notify successful submission
    notifySubmissionSuccess() {
        try {
            sessionStorage.setItem('fps_private_submitted', 'true');
            // Dispatch custom event for debugging
            window.dispatchEvent(new CustomEvent('fps:privateDataSubmitted', {
                detail: { timestamp: new Date().toISOString() }
            }));
        } catch (e) {
            // Ignore storage errors
        }
    }

    // Get public data for login integration
    async getPublicDataForLogin(timeoutMs = 3000) {
        if (!this.publicDataPromise) {
            console.warn('[FPS Core] Collection not started - initializing now');
            await this.initialize();
        }

        try {
            return await Promise.race([
                this.publicDataPromise,
                new Promise((_, reject) => 
                    setTimeout(() => reject(new Error('Timeout')), timeoutMs)
                )
            ]);
        } catch (error) {
            console.warn('[FPS Core] Public data timeout:', error);
            return this.getFallbackPublicData();
        }
    }

    // Fallback data if collection fails
    getFallbackPublicData() {
        return {
            sessionId: `fallback_${Date.now()}_${Math.random().toString(36).slice(2)}`,
            deviceUUID: this.uuidManager.generateNewUUID(),
            timestamp: new Date().toISOString(),
            basicFingerprint: {
                hardwareHash: 'fallback-hw',
                graphicsHash: 'fallback-gfx',
                audioHash: 'fallback-audio',
                browserHash: 'fallback-browser',
                apiHash: 'fallback-api'
            },
            confidence: 0.1,
            privacy: "fallback_mode"
        };
    }

    // Status check
    getStatus() {
        return {
            collectionStarted: this.collectionStarted,
            publicDataReady: !!this.publicDataPromise,
            privateDataSubmitted: this.privateDataSubmitted,
            timestamp: new Date().toISOString()
        };
    }
}

function sanitizeHeaders(headers) {
    const allowedHeaders = [
        'user-agent', 'accept-language', 'accept-encoding',
        'accept', 'referer', 'sec-fetch-site', 'sec-fetch-mode'
    ];
    
    const sanitized = {};

            }
            
            return false;

            this.submissionInProgress = false;
        }

        fp.stability_score,
        COUNT(fh.id) as fingerprint_changes,
        AVG(fh.similarity_score) as avg_similarity,
        MAX(fs.similarity_score) as max_fraud_similarity
    FROM DeviceIdentities di
    JOIN DeviceFingerprints fp ON di.current_fingerprint_id = fp.id
    LEFT JOIN DeviceFingerprintHistory fh ON di.id = fh.device_identity_id
    LEFT JOIN DeviceFingerprintSimilarity fs ON fp.id = fs.fingerprint_id_1
    WHERE fs.risk_flag = true OR fs.similarity_score > 0.85
    GROUP BY di.id, di.confidence_score, fp.stability_score
)
SELECT 
    device_identity_id,
    CASE 
        WHEN max_fraud_similarity > 0.95 THEN 85 + (fingerprint_changes * 3)
        WHEN avg_similarity > 0.8 THEN 60 + (fingerprint_changes * 5)
        WHEN confidence_score < 0.3 THEN 70 + (10 - (stability_score * 10))
        WHEN fingerprint_changes > 10 THEN 50 + fingerprint_changes
        ELSE GREATEST(20, 100 - (confidence_score * 50) - (stability_score * 30))
    END as calculated_risk_score
FROM device_metrics;

class DataLinkingService {
    constructor() {

    setupConsumers() {
        // Consumer for public data (from login)

        }, { prefetch: 50, autoAck: false });

        // Consumer for private data notifications
        this.rabbitMQ.consume('private-data-notifications', async (message) => {
            await this.handlePrivateDataNotification(message);
        }, { prefetch: 100, autoAck: false });
    }

    // Handle public data from login flow
    async handlePublicData(message) {
        const { sessionId, userId, publicData, clientIP } = JSON.parse(message.content);
        
        try {
            console.log(`[Data Linking] Processing session ${sessionId}`);
            
            // Check if private data already available
            const privateKey = `fps:private:${sessionId}`;
            const privateDataStr = await this.redis.get(privateKey);
            
            if (privateDataStr) {
                // FAST PATH: Both data available immediately
                const privateData = JSON.parse(privateDataStr);
                await this.performDataLinking(sessionId, userId, publicData, privateData, clientIP);
                
                await this.redis.del(privateKey);
                await this.rabbitMQ.ack(message);
                
                console.log(`[Data Linking] ✅ FAST PATH completed for ${sessionId}`);
            } else {
                // SLOW PATH: Store public data and wait for private
                await this.storePublicDataForLinking(sessionId, userId, publicData, clientIP);
                await this.rabbitMQ.ack(message);
                
                console.log(`[Data Linking] 🔄 SLOW PATH queued for ${sessionId}`);
            }
            
        } catch (error) {
            console.error(`[Data Linking] Failed for ${sessionId}:`, error);
            await this.rabbitMQ.nack(message, false, true);
        }
    }

    // Core data linking logic
    async performDataLinking(sessionId, userId, publicData, privateData, clientIP) {
        try {
            // Create complete linked dataset
            const linkedData = {
                sessionId, userId, clientIP,
                timestamp: new Date().toISOString(),
                
                // Public context (lightweight hashes)
                deviceUUID: publicData.deviceUUID,
                basicFingerprint: publicData.basicFingerprint,
                
                // Private details (comprehensive data)
                detailedGroups: privateData.detailedGroups,
                riskFactors: privateData.riskFactors,
                webrtcIPs: privateData.webrtcIPs,
                
                // Linking metadata
                dataCompleteness: 1.0,
                linkingMethod: 'redis_fusion'
            };

            // Store complete linked data
            const linkedRecord = await this.db.create('linked_session_data', {
                session_id: sessionId,
                user_id: userId,
                device_uuid: publicData.deviceUUID,
                client_ip: clientIP,
                linked_data: linkedData,
                data_completeness: 1.0,
                status: 'linked_ready_for_analysis',
                created_at: new Date()
            });

            // 🚀 TRIGGER ANALYSIS STREAM
            await this.rabbitMQ.publish('analysis-processing-queue', {
                linkedRecordId: linkedRecord.id,
                sessionId, userId,
                priority: 'normal',
                analysisType: 'comprehensive'
            }, { persistent: true, priority: 5 });

            console.log(`[Data Linking] ✅ Data linked and analysis triggered for ${sessionId}`);
            
        } catch (error) {
            console.error(`[Data Linking] Linking failed for ${sessionId}:`, error);
            throw error;
        }
    }

    // Store public data for later linking with retry mechanism
    async storePublicDataForLinking(sessionId, userId, publicData, clientIP) {
        const publicKey = `fps:public:${sessionId}`;
        const publicSession = {
            sessionId, userId, publicData, clientIP,
            timestamp: new Date().toISOString()
        };
        
        // Store with 30-minute TTL (longer than private data)
        await this.redis.setex(publicKey, 1800, JSON.stringify(publicSession));
        
        // Schedule smart retry with exponential backoff
        for (let i = 0; i < this.retryDelays.length; i++) {
            setTimeout(() => this.attemptRetryLinking(sessionId, i), this.retryDelays[i]);
        }
    }

    // Smart retry mechanism
    async attemptRetryLinking(sessionId, attemptNumber) {
        const publicKey = `fps:public:${sessionId}`;
        const privateKey = `fps:private:${sessionId}`;
        
        try {
            const [publicDataStr, privateDataStr] = await Promise.all([
                this.redis.get(publicKey),
                this.redis.get(privateKey)
            ]);
            
            if (publicDataStr && privateDataStr) {
                // Both available now - perform linking
                const publicSession = JSON.parse(publicDataStr);
                const privateData = JSON.parse(privateDataStr);
                
                await this.performDataLinking(
                    sessionId,
                    publicSession.userId,
                    publicSession.publicData,
                    privateData,
                    publicSession.clientIP
                );
                
                // Cleanup
                await Promise.all([
                    this.redis.del(publicKey),
                    this.redis.del(privateKey)
                ]);
                
                console.log(`[Data Linking] ✅ RETRY ${attemptNumber + 1} successful for ${sessionId}`);
                return;
            }
            
            // Final attempt - trigger basic analysis
            if (attemptNumber === this.retryDelays.length - 1 && publicDataStr) {
                const publicSession = JSON.parse(publicDataStr);
                await this.triggerBasicAnalysis(publicSession);
                await this.redis.del(publicKey);
                
                console.log(`[Data Linking] ⚠️ FALLBACK to basic analysis for ${sessionId}`);
            }
            
        } catch (error) {
            console.error(`[Data Linking] Retry ${attemptNumber + 1} failed for ${sessionId}:`, error);
        }
    }

    // Trigger basic analysis for incomplete data
    async triggerBasicAnalysis(publicSession) {
        const { sessionId, userId, publicData, clientIP } = publicSession;
        
        const basicRecord = await this.db.create('linked_session_data', {
            session_id: sessionId,
            user_id: userId,
            device_uuid: publicData.deviceUUID,
            client_ip: clientIP,
            linked_data: {
                sessionId, userId,
                basicFingerprint: publicData.basicFingerprint,
                dataCompleteness: 0.3,
                fallbackReason: 'private_data_timeout'
            },
            data_completeness: 0.3,
            status: 'basic_ready_for_analysis',
            created_at: new Date()
        });

        // Send to analysis stream with lower priority
        await this.rabbitMQ.publish('analysis-processing-queue', {
            linkedRecordId: basicRecord.id,
            sessionId, userId,
            priority: 'low',
            analysisType: 'basic_only'
        }, { persistent: true, priority: 1 });
    }

            console.log(`[Risk Analysis] Processing ${analysisType} analysis for ${sessionId}`);

            if (!linkedRecord) {

                return;
            }

            // Perform analysis based on type
            let analysisResult;
            if (analysisType === 'basic_only') {
                analysisResult = await this.performBasicAnalysis(linkedRecord);
            } else {
                analysisResult = await this.performComprehensiveAnalysis(linkedRecord);
            }

            // Store analysis result
            await this.storeAnalysisResult(linkedRecord, analysisResult, startTime);
            
            // Trigger alerts if high risk
            if (analysisResult.riskScore > 75) {
                await this.triggerHighRiskAlert(userId, sessionId, analysisResult);
            }

            await this.rabbitMQ.ack(message);
            console.log(`[Risk Analysis] ✅ ${analysisType} completed for ${sessionId}: Risk ${analysisResult.riskScore} (${Date.now() - startTime}ms)`);
            
        } catch (error) {
            console.error(`[Risk Analysis] Failed for ${sessionId}:`, error);
            await this.rabbitMQ.nack(message, false, true);
        }
    }

    // Comprehensive analysis with full data context
    async performComprehensiveAnalysis(linkedRecord) {
        const { linked_data: data, user_id: userId } = linkedRecord;
        
        try {
            // Get cached user history for context
            const userHistory = await this.getCachedUserHistory(userId, 100);
            
            // Parallel analysis of different risk factors
            const [
                deviceConsistency,
                fingerprintPatterns,
                behaviorAnalysis,
                locationConsistency,
                botDetection,
                deviceFingerprint
            ] = await Promise.all([
                this.analyzeDeviceConsistency(data, userHistory),
                this.analyzeFingerprintPatterns(data, userHistory),
                this.analyzeBehaviorPatterns(data, userHistory),
                this.analyzeLocationConsistency(data, userHistory),
                this.analyzeBotIndicators(data),
                this.generateComprehensiveFingerprint(data)
            ]);

            // Weighted risk calculation
            const riskFactors = {
                deviceConsistency,
                fingerprintPatterns,
                behaviorAnalysis,
                locationConsistency,
                botDetection
            };

            const weights = {
                deviceConsistency: 0.30,
                fingerprintPatterns: 0.25,
                behaviorAnalysis: 0.20,
                locationConsistency: 0.15,
                botDetection: 0.10
            };

            let totalRiskScore = 0;
            for (const [factor, weight] of Object.entries(weights)) {
                totalRiskScore += (riskFactors[factor] || 0) * weight * 100;
            }

            const finalRiskScore = Math.min(Math.max(Math.round(totalRiskScore), 0), 100);

            return {
                riskScore: finalRiskScore,
                riskFactors: riskFactors,
                deviceFingerprint: deviceFingerprint,
                analysisType: 'comprehensive',
                confidence: this.calculateAnalysisConfidence(riskFactors),
                dataQuality: data.dataCompleteness || 1.0,
                processingMethod: 'comprehensive_parallel_analysis'
            };
            
        } catch (error) {
            console.error('[Risk Analysis] Comprehensive analysis failed:', error);
            throw error;
        }
    }

    // Device consistency analysis with historical context
    async analyzeDeviceConsistency(data, userHistory) {
        try {
            if (!data.detailedGroups || userHistory.length === 0) {
                return 0.3; // Moderate risk for insufficient data
            }

            const currentSignature = this.extractDeviceSignature(data.detailedGroups);
            
            // Compare with recent historical signatures
            const historicalSignatures = userHistory
                .filter(entry => entry.linked_data?.detailedGroups)
                .map(entry => this.extractDeviceSignature(entry.linked_data.detailedGroups))
                .slice(0, 20); // Last 20 sessions

            if (historicalSignatures.length === 0) {
                return 0.2; // New user, low-moderate risk
            }

            // Calculate similarity scores
            const similarities = historicalSignatures.map(historical => 
                this.calculateDeviceSimilarity(currentSignature, historical)
            );

            const maxSimilarity = Math.max(...similarities);
            const avgSimilarity = similarities.reduce((a, b) => a + b, 0) / similarities.length;

            // Risk scoring based on consistency
            if (maxSimilarity > 0.95) return 0.05; // Very consistent device
            if (avgSimilarity > 0.85) return 0.15; // Consistent device
            if (avgSimilarity > 0.70) return 0.30; // Somewhat consistent
            if (avgSimilarity > 0.50) return 0.60; // Inconsistent device
            
            return 0.85; // Very inconsistent, high risk
            
        } catch (error) {
            console.error('[Risk Analysis] Device consistency analysis failed:', error);
            return 0.5; // Default moderate risk
        }
    }

            if (data.riskFactors) {
                if (data.riskFactors.browserAutomation) botScore += 0.5;
                if (data.riskFactors.headlessIndicators?.length > 0) {
                    botScore += 0.4;
                }
                if (data.riskFactors.suspiciousHeaders) botScore += 0.3;

                    if (Array.isArray(plugins) && plugins.length === 0) {
                        botScore += 0.2;

                    }

        return {
            hardware: detailedGroups.groupA_hardware?.hash,
            graphics: detailedGroups.groupB_graphics?.hash,
            audio: detailedGroups.groupC_audio?.hash,
            browser: detailedGroups.groupD_browser?.hash,
            apis: detailedGroups.groupE_apis?.hash
        };
    }

    calculateDeviceSimilarity(sig1, sig2) {
        let matches = 0;
        let total = 0;

        for (const key in sig1) {
            total++;
            if (sig1[key] === sig2[key]) {
                matches++;
            }
        }

        return total > 0 ? matches / total : 0;
    }

    generateCombinedHash(detailedGroups) {
        if (!detailedGroups) return 'unknown';
        
        const hashes = Object.values(detailedGroups)
            .map(group => group.hash)
            .filter(hash => hash)
            .join('|');
        
        return this.simpleHash(hashes);
    }

    simpleHash(str) {
        let hash = 0;
        for (let i = 0; i < str.length; i++) {
            hash = ((hash << 5) - hash + str.charCodeAt(i)) & 0xffffffff;
        }
        return Math.abs(hash).toString(16);
    }

    calculateAnalysisConfidence(riskFactors) {
        const factorCount = Object.keys(riskFactors).length;
        const validFactors = Object.values(riskFactors)
            .filter(factor => factor !== null && factor !== undefined).length;
        
        return validFactors / factorCount;
    }
